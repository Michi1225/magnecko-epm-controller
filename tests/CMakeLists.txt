cmake_minimum_required(VERSION 3.22)
project(magnecko_host_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()

add_executable(controller_logic controller_logic.cpp ../Libs/FSM/FSM.cpp)
target_include_directories(controller_logic PRIVATE ../Libs/FSM ../Libs/Controller)
add_test(NAME controller_logic COMMAND controller_logic)

add_executable(rtos_io rtos_io_test.cpp ../Libs/Runtime/rtos_io.cpp)
target_include_directories(rtos_io PRIVATE hal_shim ../Libs/Runtime)
add_test(NAME rtos_io COMMAND rtos_io)

add_executable(imu_transactions imu_test.cpp ../Libs/Sensors/BNO086.cpp)
target_include_directories(imu_transactions PRIVATE hal_shim ../Libs/Runtime ../Libs/Sensors)
add_test(NAME imu_transactions COMMAND imu_transactions)

add_executable(esc_transport esc_transport_test.c ../Libs/SOES_Wrapper/hal/esc_hw.c)
target_include_directories(esc_transport PRIVATE hal_shim ../Libs/Runtime
    ../Libs/SOES_Wrapper/hal ../Libs/SOES_Wrapper/SOES/soes
    ../Libs/SOES_Wrapper/SOES/soes/include/sys/gcc ../Libs/SOES_Wrapper/autogenerated)
target_compile_options(esc_transport PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME esc_transport COMMAND esc_transport)

add_executable(eeprom_transactions eeprom_test.cpp ../Libs/Controller/eeprom.cpp)
target_include_directories(eeprom_transactions PRIVATE hal_shim ../Libs/Runtime ../Libs/Controller)
add_test(NAME eeprom_transactions COMMAND eeprom_transactions)

foreach(target controller_logic rtos_io imu_transactions eeprom_transactions)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endforeach()
